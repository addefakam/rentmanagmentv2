    38→    38→export async function createParty(input: {
    39→    39→  type: string; fullName: string; idTypeId: string; idNumber: string;
    40→    40→  idOriginalSeen: boolean; idCopyAttached: boolean;
    41→    41→  phone?: string; address?: string; isDeaf?: boolean; usesSignLanguage?: boolean;
    42→    42→  proxyName?: string; proxyIdNumber?: string; proxyWitness1Name?: string; proxyWitness2Name?: string;
    43→    43→  registeredAtOrgUnitId: string;
    44→    44→  verifyIdentityOnline?: boolean; // Phase 5: national ID service check (Dir. Art. 7)
    45→    45→}) {
    46→    46→  requireTrue(input.idOriginalSeen && input.idCopyAttached, "Dir. Art. 7",
    47→    47→    "Identification original must be presented and a copy attached (original-and-copy rule).");
    48→    48→  if (input.type === "AGENT") {
    49→    49→    requireTrue(isProxyDocumentationComplete(input), "Dir. Art. 7",
    50→    50→      "Agent acting by proxy needs proxy identification and two witness names on the proxy documents.");
    51→    51→  }
    52→    52→  // Phase 5 integration: optional online identity verification. The service
    53→    53→  // answers MATCH / NO_MATCH / DEFERRED; a DEFERRED outcome degrades
    54→    54→  // gracefully (registration continues, manual follow-up flag), per the plan's
    55→    55→  // failure-behaviour requirement.
    56→    56→  let idCheckNote: string | null = null;
    57→    57→  if (input.verifyIdentityOnline) {
    58→    58→    const idType = await db.identificationType.findUnique({ where: { id: input.idTypeId } });
    59→    59→    const verdict = await idService.verify({
    60→    60→      idTypeCode: idType?.code ?? "UNKNOWN", idNumber: input.idNumber, fullName: input.fullName,
    61→    61→    });
    62→    62→    idCheckNote = `${verdict.status}: ${verdict.note} (${verdict.providerRef})`;
    63→    63→  }
    64→    64→  const seq = (await db.party.count()) + 1;
    65→    65→  const party = await db.party.create({
    66→    66→    data: {
    67→    67→      partyCode: `PTY-${pad(seq, 6)}`,
    68→    68→      type: input.type, fullName: input.fullName,
    69→    69→      idTypeId: input.idTypeId, idNumber: input.idNumber,
    70→    70→      idOriginalSeen: input.idOriginalSeen, idCopyAttached: input.idCopyAttached,
    71→    71→      phone: input.phone, address: input.address,
    72→    72→      isDeaf: input.isDeaf ?? false, usesSignLanguage: input.usesSignLanguage ?? false,
    73→    73→      proxyName: input.proxyName, proxyIdNumber: input.proxyIdNumber,
    74→    74→      proxyWitness1Name: input.proxyWitness1Name, proxyWitness2Name: input.proxyWitness2Name,
    75→    75→      registeredAtOrgUnitId: input.registeredAtOrgUnitId,
    76→    76→      documents: input.type === "AGENT"
    77→    77→        ? { create: { docType: "PROXY_LETTER", ref: `PROXY-${input.idNumber}`, originalSeen: true, copyAttached: true } }
    78→    78→        : undefined,
    79→    79→    },
    80→    80→  });
    81→    81→  return { ...party, idCheckNote };
    82→    82→}

   135→   135→export async function createRegistrationFile(input: {
   136→   136→  woredaId: string; propertyId: string; landlordId: string; tenantId: string; agentId?: string;
   137→   137→  modelContractId: string; monthlyRent: number; leaseStart: Date; leaseEnd: Date;
   138→   138→  prepaymentMonths: number; paymentMethod: string; paymentMethodConfirmed: boolean;
   139→   139→  interpreterUsed: boolean; interpreterName?: string; isLegacy: boolean;
   140→   140→  witnesses: { fullName: string; idTypeId: string; idNumber: string }[];
   141→   141→  enteredByOrgUnitId: string;
   142→   142→}) {
   143→   143→  // Model contract must be the active Bureau version (Proc. Art. 5; Dir. Art. 4)
   144→   144→  const mc = await db.modelContract.findUnique({ where: { id: input.modelContractId } });
   145→   145→  requireTrue(mc, "Proc. Art. 5; Dir. Art. 4", "Model contract version not found.");
   146→   146→  requireTrue(mc!.status === "ACTIVE", "Proc. Art. 5; Dir. Art. 4",
   147→   147→    "Files must be written on the ACTIVE Bureau model contract version.");
   148→   148→
   149→   149→  // Parties verified (Proc. Art. 4; Dir. Art. 7)
   150→   150→  for (const [label, id] of [["Landlord", input.landlordId], ["Tenant", input.tenantId]] as const) {
   151→   151→    const p = await db.party.findUnique({ where: { id } });
   152→   152→    requireTrue(p && p.verificationStatus === "VERIFIED", "Proc. Art. 4; Dir. Art. 7",
   153→   153→      `${label} must be a verified party before filing.`);
   154→   154→  }
   155→   155→
   156→   156→  // Statutory lease rules (Proc. Arts. 5, 6, 12, 13)
   157→   157→  const city = await db.cityConfig.findFirst({ where: { isActive: true } });
   158→   158→  const v = validateLease({
   159→   159→    leaseStart: input.leaseStart, leaseEnd: input.leaseEnd, monthlyRent: input.monthlyRent,
   160→   160→    prepaymentMonths: input.prepaymentMonths, paymentMethodConfirmed: input.paymentMethodConfirmed,
   161→   161→    minLeaseYears: city?.minLeaseYears ?? LAW.MIN_LEASE_YEARS,
   162→   162→    maxPrepayMonths: city?.maxPrepayMonths ?? LAW.MAX_PREPAYMENT_MONTHS,
   163→   163→  });
   164→   164→  if (!v.ok) throw new LegalError(v.rule, v.message);
   165→   165→
   166→   166→  // Witnesses: exactly three (model agreement)
   167→   167→  const w = validateWitnesses(input.witnesses.length);
   168→   168→  if (!w.ok) throw new LegalError(w.rule, w.message);
   169→   169→
   170→   170→  // Interpreter flow for deaf parties (Dir. Art. 8(1))
   171→   171→  const deafParty = await db.party.findFirst({
   172→   172→    where: { id: { in: [input.landlordId, input.tenantId] }, isDeaf: true },
   173→   173→  });
   174→   174→  if (deafParty) {
   175→   175→    requireTrue(input.interpreterUsed && !!input.interpreterName, "Dir. Art. 8(1)",
   176→   176→      "A deaf party requires the sign-language interpreter flow with a named interpreter.");
   177→   177→  }
   178→   178→
   179→   179→  const woreda = await db.orgUnit.findUnique({ where: { id: input.woredaId } });
   180→   180→  requireTrue(woreda && woreda.tier === "WOREDA", "Dir. Art. 6", "Filing woreda not found.");
   181→   181→  const fileNumber = await nextFileNumber(woreda!.code, input.woredaId);
   182→   182→
   183→   183→  const file = await db.registrationFile.create({
   184→   184→    data: {
   185→   185→      fileNumber, woredaId: input.woredaId,
   186→   186→      propertyId: input.propertyId, landlordId: input.landlordId, tenantId: input.tenantId,
   187→   187→      agentId: input.agentId, modelContractId: input.modelContractId,
   188→   188→      monthlyRent: input.monthlyRent, leaseStart: input.leaseStart, leaseEnd: input.leaseEnd,
   189→   189→      prepaymentMonths: input.prepaymentMonths, paymentMethod: input.paymentMethod,
   190→   190→      paymentMethodConfirmed: input.paymentMethodConfirmed,
   191→   191→      interpreterUsed: input.interpreterUsed, interpreterName: input.interpreterName,
   192→   192→      isLegacy: input.isLegacy,
   193→   193→      witnesses: { create: input.witnesses.map((wi, i) => ({ orderNo: i + 1, ...wi })) },
   194→   194→      annotations: input.isLegacy
   195→   195→        ? { create: {
   196→   196→            code: "LEGACY_ART7",
   197→   197→            text: "Pre-proclamation contract: presented within the 30+3 day window of Proc. Art. 7; annotated per Dir. Art. 8(2).",
   198→   198→            annotatedByOrgUnitId: input.enteredByOrgUnitId,
   199→   199→          } }
   200→   200→        : undefined,
   201→   201→    },
   202→   202→  });
   203→   203→
   204→   204→  if (input.isLegacy) {
   205→   205→    // Legacy 30+3 day registration clock (Proc. Art. 7; Dir. Art. 8(2))
   206→   206→    await db.deadlineTrack.create({
   207→   207→      data: {
   208→   208→        code: "LEGACY-33D", subjectType: "FILE", subjectRef: fileNumber,
   209→   209→        name: "Legacy contract registration window (30 + 3 days)",
   210→   210→        dueAt: addCalendarDays(new Date(), LAW.REGISTRATION_WINDOW_DAYS + LAW.LEGACY_GRACE_DAYS),
   211→   211→      },
   212→   212→    });
   213→   213→  }
   214→   214→  return file;
   215→   215→}

   217→   217→export async function checkChecklist(fileId: string, items: { orderNo: number; passed: boolean; note?: string }[]) {
   218→   218→  const file = await db.registrationFile.findUnique({ where: { id: fileId } });
   219→   219→  requireTrue(file, "Dir. Art. 6", "File not found.");
   220→   220→  requireTrue(file!.status === "PRESENTED" || file!.status === "CHECKLIST_PASSED", "Dir. Arts. 6-9",
   221→   221→    `Checklist can only be checked on a PRESENTED file (current: ${file!.status}).`);
   222→   222→  for (const it of items) {
   223→   223→    const tpl = NINE_POINT_CHECKLIST.find((c) => c.orderNo === it.orderNo);
   224→   224→    requireTrue(tpl, "Dir. Arts. 6-9", `Checklist item ${it.orderNo} does not exist.`);
   225→   225→    await db.registrationChecklistItem.upsert({
   226→   226→      where: { fileId_orderNo: { fileId, orderNo: it.orderNo } },
   227→   227→      create: { fileId, orderNo: it.orderNo, code: tpl!.code, requirement: tpl!.requirement, passed: it.passed, note: it.note },
   228→   228→      update: { passed: it.passed, note: it.note, checkedAt: new Date() },
   229→   229→    });
   230→   230→  }
   231→   231→  const all = await db.registrationChecklistItem.findMany({ where: { fileId } });
   232→   232→  const allPassed = NINE_POINT_CHECKLIST.every((c) => {
   233→   233→    const row = all.find((a) => a.orderNo === c.orderNo);
   234→   234→    return row?.passed === true;
   235→   235→  });
   236→   236→  if (allPassed) {
   237→   237→    return db.registrationFile.update({ where: { id: fileId }, data: { status: "CHECKLIST_PASSED" } });
   238→   238→  }
   239→   239→  return file!;
   240→   240→}

   681→   681→export async function enqueueReplication(batchRef: string, fromOrgUnitId: string, recordType: string, recordRef: string, summary?: string) {
   682→   682→  const from = await db.orgUnit.findUnique({ where: { id: fromOrgUnitId }, include: { parent: true } });
   683→   683→  requireTrue(from, "Dir. Art. 13", "Source org unit not found.");
   684→   684→  const hops: { toId: string; hopOrder: number }[] = [];
   685→   685→  let cursor = from!;
   686→   686→  let hop = 1;
   687→   687→  while (cursor.parentId && hop <= 3) {
   688→   688→    hops.push({ toId: cursor.parentId, hopOrder: hop });
   689→   689→    const parent = await db.orgUnit.findUnique({ where: { id: cursor.parentId }, include: { parent: true } });
   690→   690→    cursor = parent!;
   691→   691→    hop++;
   692→   692→  }
   693→   693→  requireTrue(hops.length > 0, "Dir. Art. 13", "Unit is already at the top of the chain (Ministry).");
   694→   694→  const rows = [];
   695→   695→  for (const h of hops) {
   696→   696→    rows.push(await db.replicationLog.create({
   697→   697→      data: { batchRef, fromOrgUnitId, toOrgUnitId: h.toId, hopOrder: h.hopOrder, recordType, recordRef, payloadSummary: summary, status: "PROPAGATED" },
   698→   698→    }));
   699→   699→  }
   700→   700→  return rows;
   701→   701→}

   720→   720→export async function computeSnapshot(orgUnitId: string, period: string) {
   721→   721→  const org = await db.orgUnit.findUnique({ where: { id: orgUnitId } });
   722→   722→  requireTrue(org, "Dir. Art. 13", "Org unit not found.");
   723→   723→  const descendantIds = await descendantOrgIds(org!);
   724→   724→  const inScope = (woredaIds: string[]) => ({ woredaId: { in: woredaIds } });
   725→   725→
   726→   726→  const contractsRegistered = await db.registrationFile.count({
   727→   727→    where: { ...inScope(descendantIds.woredaIds), status: "REGISTERED",
   728→   728→      registeredAt: { gte: new Date(`${period}-01T00:00:00Z`), lt: new Date(nextPeriodStart(period)) } },
   729→   729→  });
   730→   730→  const activeFiles = await db.registrationFile.count({
   731→   731→    where: { ...inScope(descendantIds.woredaIds), status: "REGISTERED" },
   732→   732→  });
   733→   733→  const complaintsReceived = await db.complaint.count({
   734→   734→    where: { receivedAtOrgUnitId: { in: descendantIds.allIds },
   735→   735→      receivedAt: { gte: new Date(`${period}-01T00:00:00Z`), lt: new Date(nextPeriodStart(period)) } },
   736→   736→  });
   737→   737→  const complaintsDecided = await db.complaint.count({
   738→   738→    where: { receivedAtOrgUnitId: { in: descendantIds.allIds }, status: { in: ["DECIDED", "CLOSED"] } },
   739→   739→  });
   740→   740→  const penAgg = await db.penaltyCase.aggregate({
   741→   741→    where: { propertyId: null, status: { in: ["NOTIFIED", "PAID", "REFERRED", "COURT_RECOVERY"] } },
   742→   742→    _sum: { computedAmount: true },
   743→   743→  });
   744→   744→  const penaltiesImposed = penAgg._sum.computedAmount ?? 0;
   745→   745→  return db.aggregationSnapshot.upsert({
   746→   746→    where: { orgUnitId_period: { orgUnitId, period } },
   747→   747→    create: {
   748→   748→      orgUnitId, period, sourceTier: org!.tier,
   749→   749→      contractsRegistered, activeFiles, complaintsReceived, complaintsDecided, penaltiesImposed,
   750→   750→    },
   751→   751→    update: {
   752→   752→      sourceTier: org!.tier, contractsRegistered, activeFiles,
   753→   753→      complaintsReceived, complaintsDecided, penaltiesImposed, computedAt: new Date(),
   754→   754→    },
   755→   755→  });
   756→   756→}

   758→   758→function nextPeriodStart(period: string): string {
   759→   759→  const [y, m] = period.split("-").map(Number);
   760→   760→  const ny = m === 12 ? y + 1 : y;
   761→   761→  const nm = m === 12 ? 1 : m + 1;
   762→   762→  return `${ny}-${pad(nm, 2)}-01T00:00:00Z`;
   763→   763→}

   765→   765→async function descendantOrgIds(org: { id: string; tier: string; parentId: string | null }): Promise<{ allIds: string[]; woredaIds: string[] }> {
   766→   766→  const allIds: string[] = [org.id];
   767→   767→  const woredaIds: string[] = [];
   768→   768→  if (org.tier === "WOREDA") woredaIds.push(org.id);
   769→   769→  let frontier = [org.id];
   770→   770→  for (let depth = 0; depth < 3 && frontier.length; depth++) {
   771→   771→    const children = await db.orgUnit.findMany({ where: { parentId: { in: frontier } } });
   772→   772→    frontier = [];
   773→   773→    for (const c of children) {
   774→   774→      allIds.push(c.id);
   775→   775→      if (c.tier === "WOREDA") woredaIds.push(c.id);
   776→   776→      frontier.push(c.id);
   777→   777→    }
   778→   778→  }
   779→   779→  return { allIds, woredaIds };
   780→   780→}

   786→   786→export async function amendModelContract(input: {
   787→   787→  baseContractId: string; newVersion: string; note: string;
   788→   788→  changes: { sectionCode: string; contentEn?: string; contentAm?: string; contentOm?: string }[];
   789→   789→}) {
   790→   790→  const base = await db.modelContract.findUnique({ where: { id: input.baseContractId }, include: { sections: true } });
   791→   791→  requireTrue(base, "Proc. Art. 5; Dir. Art. 4", "Base model contract not found.");
   792→   792→  requireTrue(base!.status === "ACTIVE", "Proc. Art. 5; Dir. Art. 4", "Only the ACTIVE version can be amended.");
   793→   793→  requireTrue(input.changes.length > 0, "Dir. Art. 4", "An amendment must change at least one section.");
   794→   794→  const dup = await db.modelContract.findUnique({ where: { version: input.newVersion } });
   795→   795→  requireTrue(!dup, "Dir. Art. 4", `Version ${input.newVersion} already exists.`);
   796→   796→  const next = await db.modelContract.create({
   797→   797→    data: {
   798→   798→      version: input.newVersion, status: "ACTIVE", issuedBy: base!.issuedBy,
   799→   799→      legalBasis: base!.legalBasis, effectiveFrom: new Date(),
   800→   800→      canonicalLang: base!.canonicalLang,
   801→   801→      sections: {
   802→   802→        create: base!.sections.map((s) => {
   803→   803→          const ch = input.changes.find((c) => c.sectionCode === s.code);
   804→   804→          return {
   805→   805→            orderNo: s.orderNo, code: s.code,
   806→   806→            titleEn: s.titleEn, titleAm: s.titleAm, titleOm: s.titleOm,
   807→   807→            contentEn: ch?.contentEn ?? s.contentEn,
   808→   808→            contentAm: ch?.contentAm ?? s.contentAm,
   809→   809→            contentOm: ch?.contentOm ?? s.contentOm,
   810→   810→            certificationStatus: ch ? "PENDING_LEGAL_REVIEW" : s.certificationStatus,
   811→   811→            legalBasis: s.legalBasis,
   812→   812→          };
   813→   813→        }),
   814→   814→      },
   815→   815→    },
   816→   816→  });
   817→   817→  await db.modelContract.update({ where: { id: base!.id }, data: { status: "SUPERSEDED" } });
   818→   818→  return next;
   819→   819→}

   822→   822→// M12 — Deadline sweep: mark overdue clocks (escalation duty, Dir. Art. 19)

   824→   824→export async function sweepDeadlines() {
   825→   825→  const now = new Date();
   826→   826→  const open = await db.deadlineTrack.findMany({ where: { status: "OPEN" } });
   827→   827→  let swept = 0;
   828→   828→  for (const d of open) {
   829→   829→    if (d.dueAt.getTime() < now.getTime()) {
   830→   830→      await db.deadlineTrack.update({ where: { id: d.id }, data: { status: "OVERDUE", escalated: true } });
   831→   831→      swept++;
   832→   832→    }
   833→   833→  }
   834→   834→  return { swept, openRemaining: open.length - swept };
   835→   835→}
