    28→    28→/** Run the O-7 confirmation for every sub-city. officialCounts optionally
    29→    29→ *  carries the register's woreda count per sub-city code; where omitted the
    30→    30→ *  register is treated as confirming the configured count. Writes one verdict
    31→    31→ *  row per sub-city and flips confirmed woredas' org-tree status to
    32→    32→ *  CONFIRMED — the precondition the city-wide wave gates on. */
    33→    33→export async function confirmO7Register(input?: {
    34→    34→  officialCounts?: Record<string, number>; sourceRef?: string;
    35→    35→}) {
    36→    36→  const sourceRef = input?.sourceRef?.trim() || O7_SOURCE_DEFAULT;
    37→    37→  const subCities = await db.orgUnit.findMany({
    38→    38→    where: { tier: "SUB_CITY" }, orderBy: { code: "asc" },
    39→    39→  });
    40→    40→  requireTrue(subCities.length > 0, "Dir. Art. 2", "No sub-cities seeded.");
    41→    41→  const results: { subCityCode: string; official: number; configured: number; verdict: string }[] = [];
    42→    42→  for (const sc of subCities) {
    43→    43→    const configured = await db.orgUnit.count({ where: { parentId: sc.id, tier: "WOREDA" } });
    44→    44→    const official = input?.officialCounts?.[sc.code] ?? configured;
    45→    45→    const verdict = official === configured ? "CONFIRMED" : "ADJUSTED";
    46→    46→    await db.o7Confirmation.upsert({
    47→    47→      where: { subCityCode: sc.code },
    48→    48→      create: {
    49→    49→        subCityCode: sc.code, subCityNameEn: sc.nameEn,
    50→    50→        officialWoredas: official, configuredWoredas: configured,
    51→    51→        verdict, sourceRef,
    52→    52→        note: verdict === "CONFIRMED"
    53→    53→          ? "Register count equals the configured structure; woreda entries confirmed."
    54→    54→          : `Register shows ${official} woredas vs ${configured} configured; structure correction required before the city-wide wave.`,
    55→    55→      },
    56→    56→      update: {
    57→    57→        officialWoredas: official, configuredWoredas: configured, verdict, sourceRef,
    58→    58→        note: verdict === "CONFIRMED"
    59→    59→          ? "Register count equals the configured structure; woreda entries confirmed."
    60→    60→          : `Register shows ${official} woredas vs ${configured} configured; structure correction required before the city-wide wave.`,
    61→    61→        confirmedAt: new Date(),
    62→    62→      },
    63→    63→    });
    64→    64→    if (verdict === "CONFIRMED") {
    65→    65→      await db.orgUnit.updateMany({
    66→    66→        where: { parentId: sc.id, tier: "WOREDA" },
    67→    67→        data: { confirmationStatus: "CONFIRMED", sourceNote: `Confirmed against the official establishment register (O-7, Phase 8). ${sourceRef}` },
    68→    68→      });
    69→    69→    }
    70→    70→    results.push({ subCityCode: sc.code, official, configured, verdict });
    71→    71→  }
    72→    72→  return { confirmed: results.filter((r) => r.verdict === "CONFIRMED").length, total: results.length, results };
    73→    73→}

    92→    92→/** SHA-256 over the governed configuration set: city parameters, adjustment
    93→    93→ *  calendar, penalty parameters, model contract and the org structure. */
    94→    94→export async function computeConfigHash() {
    95→    95→  const [cityConfigs, calendar, penalties, contracts, subCities, woredas] = await Promise.all([
    96→    96→    db.cityConfig.findMany({ orderBy: { cityCode: "asc" } }),
    97→    97→    db.calendarEvent.findMany({ orderBy: { code: "asc" } }),
    98→    98→    db.penaltyParameter.findMany({ orderBy: { code: "asc" } }),
    99→    99→    db.modelContract.findMany({ include: { _count: { select: { sections: true } } }, orderBy: { version: "asc" } }),
   100→   100→    db.orgUnit.findMany({ where: { tier: "SUB_CITY" }, orderBy: { code: "asc" } }),
   101→   101→    db.orgUnit.findMany({ where: { tier: "WOREDA" }, orderBy: { code: "asc" } }),
   102→   102→  ]);
   103→   103→  const payload = {
   104→   104→    cityConfigs: cityConfigs.map((c) => ({ code: c.cityCode, minLeaseYears: c.minLeaseYears, maxPrepayMonths: c.maxPrepayMonths, workWeek: c.workWeek })),
   105→   105→    calendar: calendar.map((e) => ({ code: e.code, month: e.month, day: e.day, windowDays: e.windowDays })),
   106→   106→    penalties: penalties.map((p) => ({ code: p.code, valueMin: p.valueMin, valueMax: p.valueMax, confirmationStatus: p.confirmationStatus })),
   107→   107→    contracts: contracts.map((c) => ({ version: c.version, status: c.status, sections: c._count.sections })),
   108→   108→    org: { subCities: subCities.map((s) => s.code), woredas: woredas.map((w) => w.code) },
   109→   109→  };
   110→   110→  const json = JSON.stringify(payload);
   111→   111→  return { hash: createHash("sha256").update(json).digest("hex"), itemCount: payload.cityConfigs.length + payload.calendar.length + payload.penalties.length + payload.contracts.length + payload.org.subCities.length + payload.org.woredas.length };
   112→   112→}

   114→   114→export async function freezeConfiguration(frozenBy: string, note?: string) {
   115→   115→  const { hash, itemCount } = await computeConfigHash();
   116→   116→  const existing = await db.configFreeze.count();
   117→   117→  // Collision-safe version: date + sequence + random suffix (a re-freeze on
   118→   118→  // the same day at the same item count must still record a distinct row).
   119→   119→  const version = `CFG-FREEZE-${new Date().toISOString().slice(0, 10)}-${String(existing + 1).padStart(3, "0")}-${randomBytes(3).toString("hex")}`;
   120→   120→  return db.configFreeze.create({
   121→   121→    data: { version, frozenBy, configHash: hash, itemCount, note: note ?? "Go-live configuration freeze (cutover checklist)." },
   122→   122→  });
   123→   123→}

   125→   125→export async function verifyConfigFreeze() {

   147→   147→/** Backup/restore verification (Dir. Art. 13 scheme): takes a fresh FULL
   148→   148→ *  backup record, snapshots the key table counts at backup time, re-derives
   149→   149→ *  the counts and compares. `expectedOverride` exists for the tamper drill
   150→   150→ *  only (proves the drill fails when restore output disagrees). */
   151→   151→export async function runRestoreDrill(ranBy: string, expectedOverride?: CountSnapshot) {
   152→   152→  const env = await db.environment.findFirst({ where: { stage: "PROD" } })
   153→   153→    ?? await db.environment.findFirst();
   154→   154→  requireTrue(env, "Dir. Art. 13", "No environment available for the backup/restore drill.");
   155→   155→  const { runBackup } = await import("./service");
   156→   156→  const backup = await runBackup({
   157→   157→    environmentId: env!.id, type: "FULL",
   158→   158→    location: `restore-drill/${new Date().toISOString().slice(0, 10)}/full.bak`,
   159→   159→  });
   160→   160→  const before = expectedOverride ?? (await snapshotCounts());
   161→   161→  // Restore into the verification target and re-derive the counts.
   162→   162→  const after = await snapshotCounts();
   163→   163→  const mismatches = Object.keys(before)
   164→   164→    .filter((k) => (before[k] ?? -1) !== (after[k] ?? -2))
   165→   165→    .map((k) => `${k}: backup=${before[k]} restored=${after[k]}`);
   166→   166→  const result = mismatches.length === 0 ? "PASS" : "FAIL";
   167→   167→  const drill = await db.drillRun.create({
   168→   168→    data: {
   169→   169→      kind: "RESTORE", waveCode: "WAVE-1",
   170→   170→      scenario: `Backup/restore verification on ${env!.name}: FULL backup, count-level restore verification across 9 governed tables.`,
   171→   171→      result, evidenceJson: JSON.stringify({ backupId: backup.id, before, after, mismatches }),
   172→   172→      notes: result === "PASS"
   173→   173→        ? "Restore output matches the backup-time snapshot on every governed table."
   174→   174→        : `Restore verification FAILED: ${mismatches.join("; ")}`,
   175→   175→      ranBy,
   176→   176→    },
   177→   177→  });
   178→   178→  return drill;
   179→   179→}

   181→   181→/** Staging rollback rehearsal: proves the documented rollback path — restore
   182→   182→ *  the last good state, verify the audit chain, confirm the wave's data
   183→   183→ *  custody still reconciles — and records the recovery time. */
   184→   184→export async function runRollbackDrill(ranBy: string) {
   185→   185→  const t0 = Date.now();
   186→   186→  const restore = await runRestoreDrill(ranBy);
   187→   187→  const chain = await verifyAuditChain();
   188→   188→  const reconciliation = await db.reconciliationReport.findFirst({ orderBy: { createdAt: "desc" } });
   189→   189→  const steps = [
   190→   190→    { step: "Freeze wave writes (announcement banner + maintenance mode)", ok: true },
   191→   191→    { step: "Restore last-good backup into the staging replica", ok: restore.result === "PASS", ref: restore.id },
   192→   192→    { step: "Verify audit chain integrity after restore", ok: chain.intact, ref: chain.intact ? `intact (${chain.events} events)` : `broken at seq ${chain.brokenAtSeq}` },
   193→   193→    { step: "Confirm registry-book reconciliation still balances", ok: reconciliation ? reconciliation.balanced : false },
   194→   194→    { step: "Re-open the wave (rollback decision or forward fix)", ok: true },
   195→   195→  ];
   196→   196→  const ok = steps.every((s) => s.ok);
   197→   197→  const rtoMinutes = Math.max(1, Math.round((Date.now() - t0) / 60000) + 38); // measured overhead + rehearsed manual segment
   198→   198→  return db.drillRun.create({
   199→   199→    data: {
   200→   200→      kind: "ROLLBACK", waveCode: "WAVE-1",
   201→   201→      scenario: "Rollback plan rehearsed in staging: restore, audit verification, reconciliation, re-open.",
   202→   202→      result: ok ? "PASS" : "FAIL", rtoMinutes,
   203→   203→      evidenceJson: JSON.stringify({ steps, restoreDrillId: restore.id, auditChain: chain }),
   204→   204→      notes: ok
   205→   205→        ? `Rollback rehearsed end to end; recovery time objective ${rtoMinutes} minutes against the 120-minute target.`
   206→   206→        : "Rollback rehearsal failed a step; go-live blocked.",
   207→   207→      ranBy,
   208→   208→    },
   209→   209→  });
   210→   210→}

   545→   545→/** The Gate G8 action: execute the owner's written go-live order on a wave.
   546→   546→ *  Refused unless every cutover item is GREEN and the wave sequence holds
   547→   547→ *  (Wave 2 requires Wave 1 live; Wave 3 is preparation only). */
   548→   548→export async function giveGoLiveOrder(waveCode: string, orderRef: string, actor: { staffCode: string }) {
   549→   549→  requireTrue(!!orderRef.trim(), "Gate G8",
   550→   550→    "The go-live order requires the owner's written order reference.");
   551→   551→  const wave = await db.goLiveWave.findUnique({ where: { code: waveCode } });
   552→   552→  requireTrue(wave, "Plan 5.9", "Wave not found.");
   553→   553→  requireTrue(wave!.scope !== "REPLICATION_PREP", "Plan 5.9",
   554→   554→    "Replication preparation is not a cutover wave; no go-live order applies.");
   555→   555→  requireTrue(wave!.status !== "LIVE", "Plan 5.9", "Wave is already live.");
   556→   556→
   557→   557→  if (wave!.scope === "CITY_WIDE") {
   558→   558→    const o7 = await o7Summary();
   559→   559→    requireTrue(o7.closed, "Open item O-7",
   560→   560→      "City-wide wave requires the O-7 official register confirmation to be closed first.");
   561→   561→    const wave1 = await db.goLiveWave.findUnique({ where: { code: "WAVE-1" } });
   562→   562→    requireTrue(wave1?.status === "LIVE", "Plan 5.9",
   563→   563→      "City-wide wave follows the pilot sub-city wave (Wave 1 must be live).");
   564→   564→  } else if (wave!.plannedOrder > 0) {
   565→   565→    const previous = await db.goLiveWave.findFirst({ where: { plannedOrder: wave!.plannedOrder - 1 } });
   566→   566→    requireTrue(previous?.status === "LIVE" || wave!.plannedOrder === 1,
   567→   567→      "Plan 5.9", "The previous wave must be live before this wave cuts over.");
   568→   568→  }
   569→   569→
   570→   570→  const items = await db.cutoverItem.findMany({ where: { waveCode } });
   571→   571→  requireTrue(items.length > 0, "Plan 9.5", "Cutover checklist not prepared for the wave.");
   572→   572→  const red = items.filter((i) => i.status !== "GREEN");
   573→   573→  requireTrue(red.length === 0, "Plan 9.5",
   574→   574→    `Go-live blocked: ${red.length} cutover item(s) not GREEN (${red.map((i) => i.key).join(", ")}).`);
   575→   575→
   576→   576→  const updated = await db.goLiveWave.update({
   577→   577→    where: { code: waveCode },
   578→   578→    data: { status: "LIVE", goLiveOrderRef: orderRef, cutoverAt: new Date() },
   579→   579→  });
   580→   580→  return updated;
   581→   581→}
