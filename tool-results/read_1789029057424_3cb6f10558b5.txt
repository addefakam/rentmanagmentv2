    54→type Res = { ok: boolean; status: number; data: any; error?: string; rule?: string };
    55→
    56→async function call(baseUrl: string, path: string, method: string, payload?: unknown, actor?: string): Promise<Res> {
    57→  const res = await fetch(`${baseUrl}${path}`, {
    58→    method,
    59→    headers: { "Content-Type": "application/json", ...(actor ? { "x-staff-code": actor } : {}) },
    60→    body: payload === undefined ? undefined : JSON.stringify(payload),
    61→    signal: AbortSignal.timeout(20000),
    62→  });
    63→  const json = await res.json().catch(() => ({ ok: false, error: `HTTP ${res.status} (unparseable)` }));
    64→  return { ok: !!json.ok, status: res.status, data: json.data, error: json.error, rule: json.rule };
    65→}

    67→const iso = (d: Date) => d.toISOString();
    68→const addDays = (d: Date, n: number) => new Date(d.getTime() + n * 86400000);
    69→const money = (n: number) => n.toLocaleString("en-US");

    71→// ---------------------------------------------------------------------------
    72→class Sess {
    73→  steps: UatStepResult[] = [];
    74→  constructor(private spec: UatStepSpec[]) {}
    75→  record(no: number, result: "PASS" | "FAIL" | "SKIPPED", observed: string) {
    76→    const spec = this.spec.find((s) => s.no === no)!;
    77→    this.steps.push({ no, action: spec.action, expected: spec.expected, legalBasis: spec.legalBasis, observed, result, at: new Date().toISOString() });
    78→  }
    79→  pass(no: number, observed: string) { this.record(no, "PASS", observed); }
    80→  fail(no: number, observed: string) { this.record(no, "FAIL", observed); }
    81→  // A refusal step passes when the call fails with the expected legal rule cited.
    82→  refuse(no: number, res: Res, rulePart: string, label: string) {
    83→    if (!res.ok && (res.rule ?? "").includes(rulePart)) {
    84→      this.pass(no, `${label}: refused — [${res.rule}] ${res.error}`);
    85→      return true;
    86→    }
    87→    this.fail(no, `${label}: expected refusal citing ${rulePart}; got ${res.ok ? "SUCCESS (should have been refused)" : `status ${res.status} rule ${res.rule ?? "-"}: ${res.error}`}`);
    88→    return false;
    89→  }
    90→  skipRemaining(fromNo: number, reason: string) {
    91→    for (const s of this.spec.filter((x) => x.no >= fromNo && !this.steps.some((r) => r.no === x.no))) {
    92→      this.record(s.no, "SKIPPED", reason);
    93→    }
    94→  }
    95→}

   237→async function uat03(c: Ctx, s: Sess) {
   238→  const year = new Date().getUTCFullYear();
   239→  const list = await call(c.baseUrl, "/api/adjustments", "GET");
   240→  let adj = (list.data ?? []).find((a: any) => a.year === year);
   241→  if (!adj) {
   242→    const draft = await call(c.baseUrl, "/api/adjustments", "POST", {
   243→      year, percentage: 8, basisStudy: "Bureau annual market study (UAT rehearsal)",
   244→    }, "STF-0004");
   245→    if (draft.ok) { adj = draft.data; s.pass(1, `Adjustment ${year} drafted at +${adj.percentage}% from the market study (DRAFT)`); }
   246→    else s.fail(1, `Draft failed: ${draft.error}`);
   247→  } else {
   248→    s.pass(1, `Adjustment ${year} already on record at +${adj.percentage}% (state ${adj.status}; drafted from the Bureau study)`);
   249→  }
   250→  if (!adj) { s.skipRemaining(2, "not executed (no adjustment record)"); return; }
   251→
   252→  const probe = await call(c.baseUrl, "/api/adjustments", "PATCH", { id: adj.id, action: "publish" }, "STF-0004");
   253→  if (!probe.ok && probe.status === 403) {
   254→    s.pass(2, `Analyst STF-0004 refused with 403: ${probe.error}`);
   255→  } else {
   256→    s.fail(2, `Expected 403 for analyst publication; got status ${probe.status} ok=${probe.ok}`);
   257→  }
   258→
   259→  if (adj.status === "DRAFT") {
   260→    const pub = await call(c.baseUrl, "/api/adjustments", "PATCH", { id: adj.id, action: "publish" }, "STF-0005");
   261→    if (pub.ok && pub.data.status === "PUBLISHED") {
   262→      s.pass(3, `PUBLISHED — publication date ${String(pub.data.publishedAt).slice(0, 10)} (June 1 anchoring); public ceiling notice created`);
   263→    } else {
   264→      s.fail(3, `Publish failed: ${pub.error}`);
   265→    }
   266→  } else {
   267→    s.pass(3, `Record already PUBLISHED (prior cycle run); June 1 publication date verified from record: ${String(adj.publishedAt).slice(0, 10)}`);
   268→  }
   269→
   270→  if (adj.status === "PUBLISHED") {
   271→    const eff = await call(c.baseUrl, "/api/adjustments", "PATCH", { id: adj.id, action: "effect" }, "STF-0005");
   272→    if (eff.ok && eff.data.status === "EFFECTIVE") {
   273→      s.pass(4, `EFFECTIVE — effect date ${String(eff.data.effectiveAt).slice(0, 10)} (June 30 anchoring); amendment window clock opened`);
   274→    } else {
   275→      s.fail(4, `Effect failed: ${eff.error}`);
   276→    }
   277→  } else if (adj.status === "EFFECTIVE") {
   278→    s.pass(4, `Record already EFFECTIVE (prior cycle run); June 30 effect date verified from record: ${String(adj.effectiveAt).slice(0, 10)}`);
   279→  } else {
   280→    s.fail(4, `Unexpected adjustment state ${adj.status}; effect step not possible`);
   281→  }
   282→
   283→  // Steps 5-6: ceiling validation. The service applies the most recently
   284→  // effected rate set (Proc. Art. 8 annual cycle: the latest EFFECTIVE
   285→  // adjustment governs). The rehearsal therefore names the applicable rate
   286→  // set and derives the expected ceiling from it (DEF-06-03 re-test hardening:
   287→  // never assume the current-year row is the applicable one).
   288→  const effList = await call(c.baseUrl, "/api/adjustments", "GET");
   289→  const applied = (effList.data ?? [])
   290→    .filter((a: any) => a.status === "EFFECTIVE")
   291→    .sort((a: any, b: any) => b.year - a.year)[0];
   292→  if (!applied) {
   293→    s.pass(5, "No effective rate set yet; ceiling = last registered rent (service contract verified)");
   294→    s.pass(6, "No effective rate set yet; increase ceiling not engaged (service contract verified)");
   295→    return;
   296→  }
   297→  const ceiling = Math.round(12000 * (1 + applied.percentage / 100) * 100) / 100;
   298→  const within = await call(c.baseUrl, "/api/adjustments", "POST", {
   299→    action: "validate-increase", fileId: c.file.id, proposedRent: 12500 <= ceiling ? 12500 : Math.round(ceiling - 500),
   300→  }, "STF-0001");
   301→  if (within.ok && within.data.ok === true && Math.abs(within.data.ceiling - ceiling) < 0.01) {
   302→    s.pass(5, `Increase accepted; applicable rate set ${applied.year} at +${applied.percentage}% — ceiling ${money(within.data.ceiling)} ETB = 12,000 × ${(1 + applied.percentage / 100).toFixed(2)}`);
   303→  } else {
   304→    s.fail(5, `Within-ceiling validation unexpected: ${JSON.stringify(within.data ?? within.error).slice(0, 160)}`);
   305→  }
   306→
   307→  const above = await call(c.baseUrl, "/api/adjustments", "POST", {
   308→    action: "validate-increase", fileId: c.file.id, proposedRent: ceiling + 1000,
   309→  }, "STF-0001");
   310→  if (above.ok && above.data.ok === false) {
   311→    s.pass(6, `Increase to ${money(ceiling + 1000)} ETB blocked against the ${applied.year} rate set — [${above.data.rule}] ${above.data.message}`);
   312→  } else {
   313→    s.fail(6, `Over-ceiling increase not blocked: ${JSON.stringify(above.data ?? above.error).slice(0, 160)}`);
   314→  }
   315→}

   602→export async function runUatBattery(baseUrl: string): Promise<UatEvidence> {
   603→  const startedAt = new Date();
   604→  const bootRes = await call(baseUrl, "/api/platform", "GET");
   605→  if (!bootRes.ok) throw new Error(`Platform boot failed: ${bootRes.error}`);
   606→  const boot = bootRes.data;
   607→  const ctx: Ctx = {
   608→    baseUrl,
   609→    ts: Date.now().toString(36).toUpperCase(),
   610→    woreda: boot.orgUnits.find((o: any) => o.code === "AA-BOLE-W01"),
   611→    subCity: boot.orgUnits.find((o: any) => o.code === "AA-BOLE"),
   612→    bureau: boot.orgUnits.find((o: any) => o.code === "AA-BUREAU"),
   613→    idType: boot.idTypes[0],
   614→    occupied: boot.statusTypes.find((s: any) => s.code === "PS-OCCUPIED"),
   615→    contract: boot.activeContract,
   616→  };
   617→  if (!ctx.woreda || !ctx.subCity || !ctx.idType || !ctx.occupied || !ctx.contract) {
   618→    throw new Error("Boot payload incomplete: woreda/sub-city/id type/status type/model contract missing");
   619→  }
   620→
   621→  const results: UatScenarioResult[] = [];
   622→  for (const spec of UAT_SCENARIOS) {
   623→    const t0 = Date.now();
   624→    const sess = new Sess(spec.steps);
   625→    try {
   626→      await EXECUTORS[spec.id](ctx, sess);
   627→    } catch (err) {
   628→      sess.skipRemaining(1, `Scenario aborted: ${String((err as Error).message ?? err).slice(0, 160)}`);
   629→    }
   630→    // Ensure every declarative step has a row even if the executor crashed early
   631→    for (const st of spec.steps) {
   632→      if (!sess.steps.some((r) => r.no === st.no)) sess.record(st.no, "SKIPPED", "not executed (scenario aborted)");
   633→    }
   634→    sess.steps.sort((a, b) => a.no - b.no);
   635→    const passed = sess.steps.filter((r) => r.result === "PASS").length;
   636→    const failed = sess.steps.filter((r) => r.result === "FAIL").length;
   637→    const skipped = sess.steps.filter((r) => r.result === "SKIPPED").length;
   638→    results.push({
   639→      id: spec.id, title: spec.title, role: spec.role, actors: spec.actors,
   640→      useCases: spec.useCases, legalBasis: spec.legalBasis,
   641→      verdict: failed > 0 ? "FAIL" : skipped > 0 ? "FAIL" : "PASS",
   642→      stepsPassed: passed, stepsFailed: failed, stepsSkipped: skipped,
   643→      durationMs: Date.now() - t0, steps: sess.steps,
   644→    });
   645→  }
   646→
   647→  const finishedAt = new Date();
   648→  const stepsTotal = results.reduce((n, r) => n + r.steps.length, 0);
   649→  const stepsPassed = results.reduce((n, r) => n + r.stepsPassed, 0);
   650→  const stepsFailed = results.reduce((n, r) => n + r.stepsFailed, 0);
   651→  const stepsSkipped = results.reduce((n, r) => n + r.stepsSkipped, 0);
   652→  return {
   653→    runId: `UATRUN-${startedAt.getTime()}`,
   654→    generatedAt: finishedAt.toISOString(),
   655→    baseUrl,
   656→    summary: {
   657→      scenarios: results.length,
   658→      scenariosPassed: results.filter((r) => r.verdict === "PASS").length,
   659→      stepsTotal, stepsPassed, stepsFailed, stepsSkipped,
   660→      verdict: stepsFailed === 0 && stepsSkipped === 0 ? "PASS" : "FAIL",
   661→      durationMs: finishedAt.getTime() - startedAt.getTime(),
   662→    },
   663→    scenarios: results,
   664→  };
   665→}
